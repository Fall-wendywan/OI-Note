 int ls, rs;   // 左右儿子编号

更新节点（pushUp） 
// 左右儿子更新当前节点
void pushUp(int id)
 {
 tree[id].sum = tree[tree[id].ls].sum + tree[tree[id].rs].sum;
 }
下放懒标记（pushDown） 
// 下放懒标记（id:当前点的编号，l:当前点维护的左端点，r:当前点维护的右端点）
void pushDown(int id, int l, int r)
 {
 if(tree[id].tag == -1) return;
 if(!tree[id].ls) tree[id].ls = ++tot; // 左儿子动态开点
if(!tree[id].rs) tree[id].rs = ++tot; // 右儿子动态开点
int mid = (l+r)/2;
 tree[tree[id].ls].sum = tree[id].tag*(mid-l+1);
 tree[tree[id].rs].sum = tree[id].tag*(r-mid);
 tree[tree[id].ls].tag = tree[tree[id].rs].tag = tree[id].tag;
 tree[id].tag = -1;
 }
区间更新 
// [l,r] 当前节点的区间，[x,y] 表示更新区间
void update(int &id, int l, int r, int x, int y, int k)
 {
 if(!id) id=++tot; // 动态开点
if(x <= l && r <= y) {
 tree[id].sum = k*(r-l+1);
 tree[id].tag = k;
 return;
 }
 pushDown(id,l,r);
 int mid = (l+r)/2;
 if(x <= mid) update(tree[id].ls, l, mid, x, y, k);
 if(y > mid) update(tree[id].rs, mid+1, r, x, y, k);
 pushUp(id);
 }
