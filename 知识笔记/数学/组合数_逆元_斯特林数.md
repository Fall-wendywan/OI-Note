## 组合数

1.单独求

```
int A, B, C, D;
int jc[MAXN], inv[MAXN];

    jc[0] = jc[1] = inv[0] = inv[1] = 1;//初始化
    for( int i = 2; i < MAXN; i ++ ){
        jc[i] = 1LL * jc[i - 1] * i % MOD;
        inv[i] = 1LL * ( MOD - MOD / i ) * inv[MOD % i] % MOD;

    }
    for( int i = 1; i <= MAXN; i ++ ) inv[i] = 1LL * inv[i - 1] * inv[i] % MOD;
}
int c( int a, int b ){
    return 1LL * jc[a] * inv[b] % MOD * inv[a - b] % MOD;
}

```
//下n上m
int C[MAXN * 2][MAXN * 2];
void cc(){

        for( int j = 0; j <= i; j ++ ){
            if( i == j ) C[i][j] = 1;

## 逆元

## 1.辗转相除法求逆元_欧几里得

int N, p;
//求 a ∗ x ≡ c( mod b )
//转化为 a ∗ x + b ∗ y = c 
void Exgcd( int a, int b, int &x, int &y ){
    if( !b ) x = 1, y = 0;//模数不能为0
    else Exgcd( b, a % b, y, x ), y -= a / b * x;//辗转相除
}
int main(){
    int a, x, y;
    Exgcd( a, p, x, y );

}

## 2.费马小定理求逆元+组合数

int N, M;
int jc[MAXN], inv[MAXN], zys[100], cnt[100];
int qpow( int a, int b ){
    int base = a, ans = 1;
    while( b > 0 ){
        if( b & 1 ){
            ans *= base;
            ans %= MOD;
        }
        base *= base;
        base %= MOD;
        b >>= 1;
    }
    return ans;
} 

void jiec(){
    jc[0] = jc[1] = inv[0] = inv[1] = 1;//初始化
    for( int i = 1; i < MAXN; i ++ ){
        jc[i] = 1LL * jc[i - 1] * i % MOD;
    }
    inv[0] = 1;
    inv[MAXN - 1] = qpow(jc[MAXN - 1], MOD - 2);
    for (int i = MAXN - 2; i >= 0; i--) {
        inv[i] = 1LL * inv[i + 1] * (i + 1) % MOD;
    }
}

int C( int n, int m ){
    if( n < m ) return 0;
    return jc[n] * inv[m] % MOD * inv[n - m] % MOD;
}

## 第二类斯特林数

这是第二种。
定义：把n个不同的数划分为m个集合的方案数，要求不能为空集，写作S( n, m )
划分不考虑排列次序


```
int s[MAXN][MAXN], jc[MAXN];
void pre(){
    jc[0] = jc[1] = 1;
    for( int i = 2; i < MAXN; i ++ ) jc[i] = jc[i - 1] * i % MOD;
}
void stirling(){
    memset( s, 0, sizeof( s ) );
    s[1][1] = 1;
    for( int i = 2; i <= MAXN - 1; i ++ ){
        for( int j = 1; j <= i; j ++ ){
            s[i][j] = s[i - 1][j - 1] + s[i - 1][j] * j;
            s[i][j] %= MOD;
        }
    }
}
```
