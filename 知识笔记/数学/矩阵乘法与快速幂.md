## $\textup{Part 1.}$矩阵乘法

矩阵加法、减法显然，只要两个矩阵行列相同就行。

好像就是将两个矩阵相乘的意思，但是要求第一个矩阵的列要等于第二个矩阵的行。

有矩阵 $A$ 大小为 $a\times m$  和 $B$ 大小为 $m \times b$ 。

相乘可以得到大小为 $a \times b$ 的矩阵 $C$。

<mark>注意此处行列！</mark>

 给出公式 $C_{i,j} = \sum^m_{k = 1} A_{i,k} \times B_{k, j}$。

物理意义：$C_{i, j}$ = A 的第 i 行 与 B 的第 j 列 的积

这个时候就哭呀定义结构体并且重载乘法运算符了。

结合律不会证，单位元不会证但是大致长这样

$\begin{bmatrix} 1 & 0  \\0 & 1 \end{bmatrix}$

其实就是对角线是 $1$。

此处单位元指的是一个矩阵乘上这个矩阵也不会改变其的值。

放代码：

```cpp
struct matrix{
    int a[MAXN][MAXN];
    matrix operator * ( const matrix & b ) const {
        matrix ans;
        for( int i = 1; i <= N; i ++ )
            for( int j = 1; j <= N; j ++ ){
                ans.a[i][j] = 0;
                for( int r = 1; r <= N; r ++ )
                    ans.a[i][j] = ( (long long) a[i][r] * b.a[r][j] % MOD + ans.a[i][j] % MOD ) % MOD;
            }
        return ans;
    }
}a;
```

        for( int i = 1; i <= N; i ++ )
            for( int j = 1; j <= N; j ++ )
                a[i][j] = ( i == j );

                ans.a[i][j] = 0;

            }
        return ans;
    }
    matrix qmpow( int x ){
        matrix ans, base;
        ans.init();
        base = *this;
        while( x ){
            if( x & 1 ) ans = ans * base;
            base = base * base;
            x >>= 1;
        }
        return ans;
    }
}a;

## $\textup{Part 3. Fibonacci}$

比如经典 $Fibonacci$。

考虑到 $O(N)$ 过于劣，所以使用矩阵快速幂。

定义矩阵:

$\begin{bmatrix} f_n \\ f_{n - 1} \end{bmatrix}$

然后通过将如下的上一个矩阵和神秘矩阵相乘就可以了

$\begin{bmatrix} f_{n - 1} \\ f_{n - 2} \end{bmatrix}$

神秘矩阵自己推，我好懒。

这里放个实现：

```cpp
#include<bits/stdc++.h>
#define int long long
#define IOS ios::sync_with_stdio(false);cin.tie(0);cout.tie(0)
using namespace std;
/*
link:
http://goj.wiki/d/junior/p/P1636
https://www.luogu.com.cn/problem/P1962
--------------------------------------
对于斐波那契，有特殊的矩阵乘法可以得到答案
所以我们就可以稍作推到并且利用快速幂高效的求出答案了
*/
const int MAXN = 3;
// const int p = 1e9 + 7;
int N, k, p;
int T;
struct matrix{
    int a[MAXN][MAXN];
    void init(){
        a[1][1] = a[1][2] = a[2][1] = 1;
        a[2][2] = 0;
    }
    void unit(){
        a[1][1] = 1; a[1][2] = 0;
        a[2][1] = 0; a[2][2] = 1;
    }
    matrix operator * ( const matrix & b ){
        matrix ans;
        for( int i = 1; i <= 2; i ++ )
            for( int j = 1; j <= 2; j ++ ){
                ans.a[i][j] = 0;
                for( int r = 1; r <= 2; r ++ )
                    ans.a[i][j] = ( (long long) a[i][r] * b.a[r][j] % p + ans.a[i][j] % p ) % p;
            }
        return ans;
    }
};

int mqpow( int x ){
    matrix ans, base;
    ans.unit(), base.init();
    while( x ){
        if( x & 1 ) ans = ans * base;
        base = base * base;
        x >>= 1;
    }
    return ans.a[1][1];
}
signed main(){
    IOS;
    cin >> T;
    while( T -- ){
        int kkks;
        cin >> kkks >> p;
        if( kkks == 0 || kkks == 1 ) cout << 1 % p << endl;
        else cout << mqpow( kkks ) << endl;
    }
    return 0;
}
```
